{{- /* Επιστρέφει τη λίστα άρθρων που σελιδοποιεί το _default/list.html. Κοινή πηγή για list.html και για τα meta (τίτλος/canonical σελίδων 2+), ώστε το .Paginate να καλείται πάντα με τα ίδια ορίσματα. */ -}} {{- $feed := where site.RegularPages "Type" "in" site.Params.mainSections -}} {{- if eq .Kind "term" }}{{ $feed = .Pages }}{{ end -}} {{- /* Release Status (New Features, Preview, Generally Available): ΠΑΝΤΑ ανά ημερομηνία, το νεότερο πρώτο, ανεξάρτητα από weight */ -}} {{- if and (eq .Kind "term") (eq .Data.Plural "releases") }}{{ $feed = .Pages.ByDate.Reverse }}{{ end -}} {{- /* Αρχική: σελίδες με «homeFeed: true» στο front matter (π.χ. ένα νέο εργαλείο) μπαίνουν στα «Πρόσφατα Άρθρα» με την ίδια ταξινόμηση με τα άρθρα (weight, μετά ημερομηνία). Βάλε τους και «weight: -6» για να μπουν ανάμεσα στα πρόσφατα με βάση την ημερομηνία τους. Για να φύγει, σβήνεις τη γραμμή homeFeed. */ -}} {{- if .IsHome }}{{ with where site.RegularPages "Params.homeFeed" true }}{{ $feed = (union . $feed).ByWeight }}{{ end }}{{ end -}} {{- /* Σελίδα σειράς (όχι ongoing): πρώτα τα αριθμημένα μέρη με τη σειρά ανάγνωσης, μετά τα συνοδευτικά (seriesCompanion: true) */ -}} {{- if and (eq .Kind "term") (eq .Data.Plural "series") (not .Params.ongoing) -}} {{- $parts := where .Pages.ByDate "Params.seriesCompanion" "!=" true -}} {{- $extra := where .Pages.ByDate "Params.seriesCompanion" "==" true -}} {{- $feed = union $parts $extra -}} {{- end -}} {{- return $feed -}}
